-- ---------------------------------------------------------
-- TABLE: roles
-- ---------------------------------------------------------
CREATE TABLE roles (
    role_id     SERIAL PRIMARY KEY,
    role_name   VARCHAR(50) NOT NULL UNIQUE,
    description TEXT
);

-- ---------------------------------------------------------
-- TABLE: users
-- ---------------------------------------------------------
CREATE TABLE users (
    user_id       SERIAL PRIMARY KEY,
    username      VARCHAR(50) NOT NULL,
    email         VARCHAR(255) NOT NULL UNIQUE,
    password_hash VARCHAR(255) NOT NULL,
    role_id       INT NOT NULL REFERENCES roles(role_id),
    is_active     BOOLEAN NOT NULL DEFAULT TRUE,
    created_at    TIMESTAMP NOT NULL DEFAULT NOW(),
    updated_at    TIMESTAMP NOT NULL DEFAULT NOW()
);

-- ---------------------------------------------------------
-- SEED: roles

INSERT INTO roles (role_name, description) VALUES
  ('student', 'Default account for students enrolled in courses'),
  ('instructor', 'Account for instructors who manage courses and content');

-- ---------------------------------------------------------
-- TABLE: classrooms
-- ---------------------------------------------------------
CREATE TABLE classrooms (
    classroom_id    SERIAL PRIMARY KEY,
    teacher_id      INT NOT NULL REFERENCES users(user_id) ON DELETE CASCADE,
    classroom_name  VARCHAR(100) NOT NULL,
    join_code       VARCHAR(10) NOT NULL UNIQUE,     
    school_year     VARCHAR(20) NOT NULL,    
    is_active       BOOLEAN NOT NULL DEFAULT TRUE,
    created_at      TIMESTAMP NOT NULL DEFAULT NOW()
);

-- ---------------------------------------------------------
-- TABLE: classroom_students
-- ---------------------------------------------------------
CREATE TABLE classroom_students (
    classroom_id    INT NOT NULL REFERENCES classrooms(classroom_id) ON DELETE CASCADE,
    student_id      INT NOT NULL REFERENCES users(user_id) ON DELETE CASCADE,
    enrolled_at     TIMESTAMP NOT NULL DEFAULT NOW(),
    PRIMARY KEY (classroom_id, student_id)
);
