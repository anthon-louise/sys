-- ---------------------------------------------------------
-- TABLE: roles
-- ---------------------------------------------------------
CREATE TABLE roles (
    role_id     SERIAL PRIMARY KEY,
    role_name   VARCHAR(50) NOT NULL UNIQUE,
    description TEXT
);

-- ---------------------------------------------------------
-- TABLE: users
-- ---------------------------------------------------------
CREATE TABLE users (
    user_id       SERIAL PRIMARY KEY,
    username      VARCHAR(50) NOT NULL,
    email         VARCHAR(255) NOT NULL UNIQUE,
    password_hash VARCHAR(255) NOT NULL,
    role_id       INT NOT NULL REFERENCES roles(role_id),
    is_active     BOOLEAN NOT NULL DEFAULT TRUE,
    created_at    TIMESTAMP NOT NULL DEFAULT NOW(),
    updated_at    TIMESTAMP NOT NULL DEFAULT NOW()
);

-- ---------------------------------------------------------
-- TABLE: classrooms
-- ---------------------------------------------------------
CREATE TABLE classrooms (
    classroom_id    SERIAL PRIMARY KEY,
    teacher_id      INT NOT NULL REFERENCES users(user_id) ON DELETE CASCADE,
    classroom_name  VARCHAR(100) NOT NULL,
    join_code       VARCHAR(10) NOT NULL UNIQUE,     
    school_year     VARCHAR(20) NOT NULL,    
    is_active       BOOLEAN NOT NULL DEFAULT TRUE,
    created_at      TIMESTAMP NOT NULL DEFAULT NOW()
);

-- ---------------------------------------------------------
-- TABLE: classroom_students
-- ---------------------------------------------------------
CREATE TABLE classroom_students (
    classroom_id    INT NOT NULL REFERENCES classrooms(classroom_id) ON DELETE CASCADE,
    student_id      INT NOT NULL REFERENCES users(user_id) ON DELETE CASCADE,
    enrolled_at     TIMESTAMP NOT NULL DEFAULT NOW(),
    PRIMARY KEY (classroom_id, student_id)
);


-- ---------------------------------------------------------
-- TABLE: problems
-- ---------------------------------------------------------
CREATE TABLE problems (
    problem_id      SERIAL PRIMARY KEY,
    teacher_id      INT NOT NULL REFERENCES users(user_id) ON DELETE CASCADE,
    title           VARCHAR(100) NOT NULL,
    description     TEXT,
    language        VARCHAR(50) NOT NULL,       -- e.g., 'C', 'Java', 'Python'
    difficulty      VARCHAR(20),                -- e.g., 'Easy', 'Medium', 'Hard'
    starter_code    TEXT,
    created_at      TIMESTAMP NOT NULL DEFAULT NOW()
);

CREATE TABLE test_cases (
    test_case_id    SERIAL PRIMARY KEY,
    problem_id      INT NOT NULL REFERENCES problems(problem_id) ON DELETE CASCADE,
    input_data      TEXT,                       -- The stdin sent to Judge0
    expected_output TEXT,                       -- The expected stdout from Judge0
    is_hidden       BOOLEAN NOT NULL DEFAULT FALSE, -- If true, students can't see it
    order_index     INT NOT NULL DEFAULT 0      -- To keep test cases in a specific order
);



-- ---------------------------------------------------------
-- TABLE: working_solutions
-- Stores the teacher's correct solution code, used to 
-- validate that the test cases are accurate before publishing.
-- ---------------------------------------------------------
CREATE TABLE working_solutions (
    solution_id     SERIAL PRIMARY KEY,
    problem_id      INT NOT NULL REFERENCES problems(problem_id) ON DELETE CASCADE,
    source_code     TEXT NOT NULL,              -- The teacher's correct code
    language        VARCHAR(50) NOT NULL,       -- Language of this specific solution (C, Java, Python)
    validated_by    INT REFERENCES users(user_id), -- The teacher who ran the validation
    validated_at    TIMESTAMP                   -- NULL until validated via Judge0
);


-- ---------------------------------------------------------
-- TABLE: assessments 
-- Stores the programming assessments assigned by teachers.
-- ---------------------------------------------------------
CREATE TABLE assessments (
    assessment_id       SERIAL PRIMARY KEY,
    classroom_id        INT NOT NULL REFERENCES classrooms(classroom_id) ON DELETE CASCADE,
    teacher_id          INT NOT NULL REFERENCES users(user_id) ON DELETE CASCADE,
    title               VARCHAR(100) NOT NULL,
    description         TEXT,
    assessment_type     VARCHAR(20) NOT NULL,       -- 'Practice', 'Quiz', or 'Exam'
    academic_term       VARCHAR(20) NOT NULL,       -- 'Midterm' or 'Finals'
    time_limit_minutes  INT,                        -- NULL means no time limit
    created_at          TIMESTAMP NOT NULL DEFAULT NOW(),
    updated_at          TIMESTAMP NOT NULL DEFAULT NOW()
);

-- ---------------------------------------------------------
-- TABLE: assessment_problems (Table 9)
-- Links problems from the teacher's bank to a specific assessment.
-- ---------------------------------------------------------
CREATE TABLE assessment_problems (
    assessment_id   INT NOT NULL REFERENCES assessments(assessment_id) ON DELETE CASCADE,
    problem_id      INT NOT NULL REFERENCES problems(problem_id) ON DELETE CASCADE,
    problem_order   INT NOT NULL,                   -- The order the problems appear in the assessment
    PRIMARY KEY (assessment_id, problem_id)         -- Prevents adding the same problem twice
);





-- ---------------------------------------------------------
-- TABLE: submissions
-- Stores the actual code a student submitted for a specific 
-- problem within an assessment.
-- ---------------------------------------------------------
CREATE TABLE submissions (
    submission_id       SERIAL PRIMARY KEY,
    assessment_id       INT NOT NULL REFERENCES assessments(assessment_id) ON DELETE CASCADE,
    student_id          INT NOT NULL REFERENCES users(user_id) ON DELETE CASCADE,
    problem_id          INT NOT NULL REFERENCES problems(problem_id) ON DELETE CASCADE,
    source_code         TEXT NOT NULL,              -- The student's code
    language            VARCHAR(50) NOT NULL,       -- Language used (C, Java, Python)
    score               DECIMAL(5,2) NOT NULL DEFAULT 0, -- Percentage or points (e.g., 85.50)
    status              VARCHAR(20) NOT NULL,       -- 'Correct', 'Incorrect', 'Partial', 'Error'
    execution_time_ms   INT,                        -- Total time taken across all test cases
    memory_used_kb      INT,                        -- Max memory used across all test cases
    submitted_at        TIMESTAMP NOT NULL DEFAULT NOW(),
    
    -- Prevents a student from having multiple submission rows for the exact same problem/assessment
    UNIQUE(assessment_id, student_id, problem_id) 
);

-- ---------------------------------------------------------
-- TABLE: submission_test_results
-- Stores the detailed output of every test case run against 
-- a student's submission. Used to show feedback.
-- ---------------------------------------------------------
CREATE TABLE submission_test_results (
    result_id           SERIAL PRIMARY KEY,
    submission_id       INT NOT NULL REFERENCES submissions(submission_id) ON DELETE CASCADE,
    test_case_id        INT NOT NULL REFERENCES test_cases(test_case_id) ON DELETE CASCADE,
    actual_output       TEXT,                       -- What the student's code actually printed
    passed              BOOLEAN NOT NULL DEFAULT FALSE, -- Did it match expected_output?
    execution_time_ms   INT,                        -- Time for this specific test case
    error_type          VARCHAR(50),                -- 'None', 'Compilation Error', 'Runtime Error', 'Time Limit', 'Wrong Answer'
    
    UNIQUE(submission_id, test_case_id)
);


-- ---------------------------------------------------------
-- TABLE: focus_loss_logs
-- Records every time a student leaves or switches away from 
-- the assessment window during an active session.
-- ---------------------------------------------------------
CREATE TABLE focus_loss_logs (
    log_id              SERIAL PRIMARY KEY,
    assessment_id       INT NOT NULL REFERENCES assessments(assessment_id) ON DELETE CASCADE,
    student_id          INT NOT NULL REFERENCES users(user_id) ON DELETE CASCADE,
    event_type          VARCHAR(50) NOT NULL,       -- e.g., 'blur', 'visibilitychange'
    occurred_at         TIMESTAMP NOT NULL DEFAULT NOW(),
    duration_seconds    INT NOT NULL DEFAULT 0      -- How long they were away
);
